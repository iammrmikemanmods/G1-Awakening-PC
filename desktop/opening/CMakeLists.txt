cmake_minimum_required(VERSION 3.20)
project(G1Opening LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
enable_testing()
add_executable(startup_contract_tests ../../tests/startup_contract_tests.cpp)
target_include_directories(startup_contract_tests PRIVATE .)
add_test(NAME startup_contract COMMAND startup_contract_tests)
if(WIN32)
  add_executable(G1OpeningPlayer WIN32 g1_opening_player.cpp)
  target_compile_definitions(G1OpeningPlayer PRIVATE UNICODE _UNICODE)
  target_link_libraries(G1OpeningPlayer PRIVATE mfplay ole32 shell32 user32 gdi32)
  if(MINGW)
    target_link_options(G1OpeningPlayer PRIVATE -municode)
  endif()
endif()
if(MSVC)
  target_compile_options(startup_contract_tests PRIVATE /W4 /WX)
  if(TARGET G1OpeningPlayer)
    target_compile_options(G1OpeningPlayer PRIVATE /W4 /WX)
    set_property(TARGET G1OpeningPlayer PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  endif()
else()
  target_compile_options(startup_contract_tests PRIVATE -Wall -Wextra -Werror -pedantic)
endif()
